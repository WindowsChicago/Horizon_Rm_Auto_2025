# CMake 最低版本要求
cmake_minimum_required(VERSION 3.16)

project(rm_auto)

#add_definitions(-std=c++11)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE "Release")
#set(CMAKE_BUILD_TYPE "debug")
set(CMAKE_MODULE_PATH ${CMAKE_CURRENT_SOURCE_DIR})
FIND_PROGRAM(CCACHE_FOUND ccache) # ccache编译器缓存（NT内核必须开启）
IF (CCACHE_FOUND)
    SET_PROPERTY(GLOBAL PROPERTY RULE_LAUNCH_COMPILE ccache)
    SET_PROPERTY(GLOBAL PROPERTY RULE_LAUNCH_LINK ccache)
    MESSAGE("< Use ccache for compiler >")
ENDIF ()


set(CMAKE_EXE_LINKER_FLAGS "${CMAKE_EXE_LINKER_FLAGS} -Wl,-z,stack-size=1600000000")

# 各个相机库文件的放置目录
set(MV_SDK /usr/lib/libMVSDK.so) #迈德

#SET(YAML_CPP_LIBRARIES /usr/local/lib/libyaml-cpp.so) # 设置yaml-cpp库，下面已经有了所以这里不用了

# C++11 标准、o3优化、多线程（NT内核必须开启）

SET(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} -O2 -std=c++17 -lstdc++ -Wall -g2 -ggdb -pthread -fopenmp -g -Wfatal-errors -D_MWAITXINTRIN_H_INCLUDED") # 定义Debug编译参数
