SET(CMAKE_CXX_FLAGS_DEBUG "$ENV{CXXFLAGS} -O2 -std=c++14 -lstdc++ -Wall -g2 -ggdb -pthread -fopenmp -g -Wfatal-errors -D_MWAITXINTRIN_H_INCLUDED") # 定义Debug编译参数
